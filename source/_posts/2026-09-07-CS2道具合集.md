<script>
document.addEventListener('DOMContentLoaded', function () {
  document.querySelectorAll('.cs2-filter').forEach(function (box) {
    var selects = box.querySelectorAll('select');
    var throwSel = selects[0], targetSel = selects[1];

    // 扫描本地图内容：到下一个 h1/h2/h3 为止；h4 记为地点分组
    var items = [], sections = [], cur = null;
    var node = box.nextElementSibling;
    while (node && !/^H[123]$/.test(node.tagName)) {
      if (node.tagName === 'H4') {
        cur = { el: node, items: [] };
        sections.push(cur);
      } else if (node.tagName === 'DETAILS') {
        var parts = node.querySelector('summary').textContent
          .split('-').map(function (s) { return s.trim(); });
        if (parts.length >= 3) {
          var it = { el: node, throwLoc: parts[0], target: parts[1], section: cur };
          items.push(it);
          if (cur) cur.items.push(it);
        }
      }
      node = node.nextElementSibling;
    }

    // 填充下拉框（去重 + 排序）
    function fill(sel, key) {
      Array.from(new Set(items.map(function (i) { return i[key]; }))).sort()
        .forEach(function (v) {
          var op = document.createElement('option');
          op.value = v; op.textContent = v;
          sel.appendChild(op);
        });
    }
    fill(throwSel, 'throwLoc');
    fill(targetSel, 'target');

    function render() {
      var t = throwSel.value, g = targetSel.value;
      var active = t || g;   // 有任一条件即进入筛选模式
      items.forEach(function (it) {
        var show = !active || ((!t || it.throwLoc === t) && (!g || it.target === g));
        it.el.style.display = show ? '' : 'none';
        if (active && show) it.el.setAttribute('open', '');       // 匹配项直接展开
        else it.el.removeAttribute('open');                        // 退出筛选时全部收起
      });
      // 地点小标题（h4）下若没有匹配项，也一并隐藏
      sections.forEach(function (sec) {
        var any = sec.items.some(function (it) { return it.el.style.display !== 'none'; });
        sec.el.style.display = (any || !active) ? '' : 'none';
      });
    }

    var resetBtn = box.querySelector('.cs2-reset');
    if (resetBtn) resetBtn.onclick = function () {
      throwSel.value = '';
      targetSel.value = '';
      render();
    };

    throwSel.onchange = render;
    targetSel.onchange = render;
  });
});
</script>
